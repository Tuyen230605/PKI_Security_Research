\resizebox{0.86\textwidth}{!}{%
\begin{tikzpicture}[
    % Định dạng hộp đen trắng với các cấp độ sắc xám học thuật
    rootbox/.style={draw=black, very thick, double, fill=gray!18, rounded corners=6pt, text width=6.6cm, align=center, inner sep=6pt, font=\small, text=black},
    cabox/.style={draw=black, thick, fill=gray!10, rounded corners=6pt, text width=5.8cm, align=center, inner sep=6pt, font=\small, text=black},
    rabox/.style={draw=black, thick, fill=gray!5, rounded corners=6pt, text width=3.4cm, align=center, inner sep=6pt, font=\small, text=black},
    repobox/.style={draw=black, thick, fill=gray!5, rounded corners=6pt, text width=3.6cm, align=center, inner sep=6pt, font=\small, text=black},
    leafbox/.style={draw=black, thick, fill=white, rounded corners=6pt, text width=6.0cm, align=center, inner sep=6pt, font=\small, text=black},
    rpbox/.style={draw=black, thick, fill=gray!12, rounded corners=6pt, text width=6.8cm, align=center, inner sep=6pt, font=\small, text=black},
    % Kiểu mũi tên phân biệt bằng nét liền, nét đứt và nét kép
    arrow/.style={-{Stealth[scale=1.1]}, thick, black},
    dashedarrow/.style={-{Stealth[scale=1.05]}, thick, dashed, black},
    trustarrow/.style={-{Stealth[scale=1.15]}, very thick, dash pattern=on 6pt off 3pt, black}
]
    % --- TẦNG 1: Root CA (Đỉnh trên cùng - Viền kép Gốc tin cậy) ---
    \node[rootbox] (root) at (9.0, 15.0) {
        \textbf{Tổ chức chứng thực gốc (Root CA)}\\[2pt]
        {\scriptsize Khóa $SK_{\mathrm{Root}}$ giữ Ngoại tuyến\\(Offline tuyệt đối trong thiết bị HSM)}
    };

    % --- TẦNG 2: Intermediate CA (Sub-CA) ---
    \node[cabox] (subca) at (9.0, 11.2) {
        \textbf{Intermediate CA (Sub-CA)}\\[2pt]
        {\scriptsize Khóa $(SK_{\mathrm{Sub}}, PK_{\mathrm{Sub}})$ Trực tuyến (Online),\\ký phát hành chứng thư máy chủ TLS}
    };

    % --- TẦNG 3: RA (Trái) và Kho lưu trữ / OCSP (Phải) ---
    \node[rabox] (ra) at (1.8, 7.0) {
        \textbf{Cơ quan đăng ký (RA)}\\[2pt]
        {\scriptsize Thẩm tra định danh,\\tiếp nhận \& duyệt CSR}
    };
    
    \node[repobox] (repo) at (16.2, 7.0) {
        \textbf{Kho lưu trữ \& OCSP}\\[2pt]
        {\scriptsize Phân phối danh sách CRL,\\phản hồi OCSP trực tuyến}
    };

    % --- TẦNG 4: Chủ thể - Web Server ---
    \node[leafbox] (leaf) at (9.0, 3.2) {
        \textbf{Chủ thể -- Web Server (Subject)}\\[2pt]
        {\scriptsize Sở hữu cặp khóa $(SK_{\mathrm{Server}}, PK_{\mathrm{Server}})$\\Thực thể cuối: End-Entity ($cA = \text{FALSE}$)}
    };

    % --- TẦNG 5: Bên tin cậy - Trình duyệt / Client ---
    \node[rpbox] (rp) at (9.0, -0.6) {
        \textbf{Bên tin cậy -- Client / Trình duyệt (Relying Party)}\\[2pt]
        {\scriptsize Sở hữu kho Root Trust Store nội bộ,\\thẩm định chuỗi chứng thư và kiểm tra trạng thái thu hồi}
    };

    % ==============================================================
    % CÁC LUỒNG TƯƠNG TÁC (Đen trắng phân định bằng kiểu nét vẽ)
    % ==============================================================

    % 1. Root CA ký cấp ủy quyền cho Sub-CA
    \draw[arrow] (root.south) -- (subca.north)
        node[midway, right=8pt, font=\small\bfseries, text=black] {1. Ký ủy quyền Sub-CA};

    % 2. Chủ thể nộp CSR cho RA
    \draw[dashedarrow] (leaf.west) -| (ra.south);
    \node[text=black, font=\small\bfseries, above=4pt] at (3.9, 3.2) {2. Gửi yêu cầu CSR};

    % 3. RA duyệt CSR chuyển sang Sub-CA
    \draw[dashedarrow] (ra.north) |- (subca.west);
    \node[text=black, font=\small\bfseries, above=4pt] at (3.95, 11.2) {3. Phê duyệt CSR};

    % 4. Sub-CA ký phát hành chứng thư cho Chủ thể
    \draw[arrow] (subca.south) -- (leaf.north)
        node[midway, right=8pt, font=\small\bfseries, align=left, text=black] {4. Ký cấp chứng thư\\TLS cho Chủ thể};

    % Xuất bản CRL: Từ Sub-CA sang đỉnh Kho lưu trữ
    \draw[arrow, dash pattern=on 4pt off 2pt] (subca.east) -| (repo.north);
    \node[text=black, font=\small\bfseries, above=4pt] at (14.0, 11.2) {Xuất bản CRL};

    % 5. Chủ thể trình chứng thư số trong bắt tay TLS cho Client
    \draw[arrow] (leaf.south) -- (rp.north)
        node[midway, right=8pt, font=\small\bfseries, text=black] {5. Trình chứng thư (TLS Handshake)};

    % 6. Client truy vấn kiểm tra thu hồi
    \draw[dashedarrow] (rp.east) -| (repo.south);
    \node[text=black, font=\small\bfseries, align=right, left=10pt] at (16.2, 2.5) {6. Truy vấn\\OCSP / CRL};

    % 7. Chuỗi tin cậy: Relying Party neo niềm tin vào Root CA qua Trust Store (nét đứt đậm)
    \draw[trustarrow, rounded corners=12pt] (rp.west) -- (-1.2, -0.6) -- (-1.2, 15.0) -- (root.west);
    \node[text=black, font=\scriptsize, align=left, right=8pt] at (-1.2, 12.8) {
        \textbf{Gốc tin cậy (Trust Anchor):}\\
        Client lưu sẵn khóa Root CA\\
        trong Root Trust Store nội bộ
    };

\end{tikzpicture}%
}
